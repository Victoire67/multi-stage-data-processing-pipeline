1"""Stage 1: normalise raw text so it can be analysed reliably.""" 

2 

3from __future__ import annotations 

4 

5import html 

6import re 

7import unicodedata 

8 

9from text_pipeline.exceptions import CleaningError 

10from text_pipeline.models import CleanedText 

11from text_pipeline.stages.base import PipelineStage 

12 

13_HTML_TAG = re.compile(r"<[^>]+>") 

14_URL = re.compile(r"(?:https?://|www\.)\S+", re.IGNORECASE) 

15# Anything that is not a letter/digit, whitespace or an apostrophe. Apostrophes 

16# must survive cleaning: the analyzer relies on negators such as "don't". 

17_NON_WORD = re.compile(r"[^\w\s']|_") 

18_WHITESPACE = re.compile(r"\s+") 

19 

20 

21class TextCleaner(PipelineStage[str, CleanedText]): 

22 """Turn raw user text into normalised, tokenised text. 

23 

24 Cleaning steps, in order: 

25 

26 1. Unicode NFKC normalisation (e.g. full-width letters become ASCII). 

27 2. Unescape HTML entities and strip HTML tags. 

28 3. Remove URLs. 

29 4. Optionally lower-case. 

30 5. Replace punctuation/symbols with spaces (apostrophes inside words such as 

31 ``don't`` are kept because the analyzer uses them for negation). 

32 6. Collapse whitespace and split into tokens. 

33 

34 The cleaner has a single responsibility: it never scores or stores text. 

35 """ 

36 

37 name = "cleaner" 

38 error_type = CleaningError 

39 

40 def __init__(self, *, lowercase: bool = True, max_length: int = 10_000) -> None: 

41 """Configure the cleaner. 

42 

43 Args: 

44 lowercase: Lower-case the text when ``True``. 

45 max_length: Maximum accepted raw input length, in characters. 

46 

47 Raises: 

48 ValueError: If ``max_length`` is not positive. 

49 """ 

50 if max_length <= 0: 

51 raise ValueError("max_length must be a positive integer") 

52 self.lowercase = lowercase 

53 self.max_length = max_length 

54 

55 def transform(self, data: str) -> CleanedText: 

56 """Clean ``data``. 

57 

58 Args: 

59 data: Raw input text. 

60 

61 Returns: 

62 The cleaned and tokenised text. 

63 

64 Raises: 

65 CleaningError: If the input is not a string, is empty, is too long 

66 or contains nothing meaningful after cleaning. 

67 """ 

68 if not isinstance(data, str): 

69 raise CleaningError(f"expected str, got {type(data).__name__}") 

70 if not data.strip(): 

71 raise CleaningError("input text is empty") 

72 if len(data) > self.max_length: 

73 raise CleaningError( 

74 f"input of {len(data)} characters exceeds maximum length " 

75 f"of {self.max_length}" 

76 ) 

77 

78 text = unicodedata.normalize("NFKC", data) 

79 text = html.unescape(text) 

80 text = _HTML_TAG.sub(" ", text) 

81 text = _URL.sub(" ", text) 

82 if self.lowercase: 

83 text = text.lower() 

84 text = _NON_WORD.sub(" ", text) 

85 

86 tokens = tuple( 

87 token.strip("'") for token in _WHITESPACE.split(text) if token.strip("'") 

88 ) 

89 if not tokens: 

90 raise CleaningError("no meaningful content left after cleaning") 

91 

92 return CleanedText(original=data, cleaned=" ".join(tokens), tokens=tokens)
